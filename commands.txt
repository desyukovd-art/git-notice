git init — создание нового репозитория
git status — проверка состояния репозитория
git add — добавление файлов в индекс
git commit — сохранение изменений
git log — просмотр истории коммитов
git branch — просмотр и создание веток
git switch — переключение между ветками
git merge — объединение веток
git clone — клонирование репозитория
git push — отправка изменений на GitHub
git pull — получение изменений с GitHub
git fetch — получение информации об изменениях
git diff — просмотр различий между версиями
git rm — удаление файла из Git
git restore — восстановление изменений
